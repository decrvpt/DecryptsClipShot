namespace DecryptsClipShot;

/// <summary>
/// ClipShot's online side: Discord sign-in (through Supabase), friends and shared clips.
///
/// Sign-in opens Discord in the browser. Supabase sends the browser back to a
/// one-time address on this PC (http://127.0.0.1:53682/callback), where ClipShot
/// picks up a code and swaps it for a session (the PKCE flow: no secret is
/// stored in the app). The session is saved encrypted for this Windows account
/// only (DPAPI) and refreshed automatically, so signing in is needed once.
///
/// The project address and public key come from cloud.json next to the app.
/// Without them the online features are switched off and no sign-in is asked.
/// </summary>
public sealed class CloudService
{
    public const int CallbackPort = 53682;
    public static string CallbackUrl => $"http://127.0.0.1:{CallbackPort}/callback";

    private static readonly HttpClient Http = new() { Timeout = TimeSpan.FromMinutes(30) };
    private readonly string _url;
    private readonly string _key;
    private readonly string _sessionPath;
    private Session? _session;

    public static CloudService Instance { get; } = Load();

    public bool Enabled => _url.Length > 0 && _key.Length > 0;
    public bool SignedIn => _session is not null;
    public string? UserId => _session?.UserId;
    public string Username => _session?.Username ?? "";
    public string? AvatarUrl => _session?.AvatarUrl;

    private CloudService(string url, string key, string sessionPath)
    {
        _url = url.TrimEnd('/');
        _key = key;
        _sessionPath = sessionPath;
        _session = ReadSession();
    }

    private static CloudService Load()
    {
        string url = "", key = "";
        try
        {
            string path = Path.Combine(AppContext.BaseDirectory, "cloud.json");
            if (File.Exists(path))
            {
                using JsonDocument document = JsonDocument.Parse(File.ReadAllText(path));
                if (document.RootElement.TryGetProperty("supabaseUrl", out JsonElement u)) url = u.GetString() ?? "";
                if (document.RootElement.TryGetProperty("publishableKey", out JsonElement k)) key = k.GetString() ?? "";
            }
        }
        catch (Exception exception) when (exception is IOException or JsonException or UnauthorizedAccessException) { }
        string data = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "Decrypts ClipShot");
        return new CloudService(url.Trim(), key.Trim(), Path.Combine(data, "session.dat"));
    }

    // ---------- Session storage (encrypted for this Windows user) ----------

    private sealed record Session(string AccessToken, string RefreshToken, DateTimeOffset ExpiresAt,
                                  string UserId, string Username, string? AvatarUrl);

    private Session? ReadSession()
    {
        try
        {
            if (!File.Exists(_sessionPath)) return null;
            byte[] plain = ProtectedData.Unprotect(File.ReadAllBytes(_sessionPath), null, DataProtectionScope.CurrentUser);
            return JsonSerializer.Deserialize<Session>(plain);
        }
        catch (Exception exception) when (exception is IOException or CryptographicException or JsonException or UnauthorizedAccessException)
        {
            return null;
        }
    }

    private void SaveSession(Session? session)
    {
        _session = session;
        try
        {
            if (session is null)
            {
                if (File.Exists(_sessionPath)) File.Delete(_sessionPath);
                return;
            }
            Directory.CreateDirectory(Path.GetDirectoryName(_sessionPath)!);
            byte[] protectedBytes = ProtectedData.Protect(JsonSerializer.SerializeToUtf8Bytes(session), null, DataProtectionScope.CurrentUser);
            File.WriteAllBytes(_sessionPath, protectedBytes);
        }
        catch (Exception exception) when (exception is IOException or CryptographicException or UnauthorizedAccessException) { }
    }

    public void SignOut() => SaveSession(null);

    // ---------- Discord sign-in ----------

    /// <summary>Opens Discord sign-in in the browser and waits for it to come back.</summary>
    public async Task SignInWithDiscordAsync(CancellationToken cancel)
    {
        string verifier = Base64Url(RandomNumberGenerator.GetBytes(48));
        string challenge = Base64Url(SHA256.HashData(Encoding.ASCII.GetBytes(verifier)));
        string authorize = $"{_url}/auth/v1/authorize?provider=discord" +
                           $"&redirect_to={Uri.EscapeDataString(CallbackUrl)}" +
                           $"&code_challenge={challenge}&code_challenge_method=s256" +
                           "&scopes=identify";

        using var listener = new HttpListener();
        listener.Prefixes.Add($"http://127.0.0.1:{CallbackPort}/");
        try
        {
            listener.Start();
        }
        catch (HttpListenerException)
        {
            throw new CloudException("ClipShot couldn't get sign-in ready. Close other copies of ClipShot and try again.");
        }
        Process.Start(new ProcessStartInfo(authorize) { UseShellExecute = true });

        string? code = null, error = null;
        using (cancel.Register(() => { try { listener.Stop(); } catch { } }))
        {
            while (code is null && error is null)
            {
                HttpListenerContext context;
                try { context = await listener.GetContextAsync(); }
                catch (Exception) when (cancel.IsCancellationRequested) { throw new OperationCanceledException(cancel); }
                if (context.Request.Url?.AbsolutePath.TrimEnd('/') != "/callback")
                {
                    context.Response.StatusCode = 404;
                    context.Response.Close();
                    continue;
                }
                code = context.Request.QueryString["code"];
                error = code is null ? context.Request.QueryString["error_description"] ?? context.Request.QueryString["error"] ?? "Sign-in was cancelled." : null;
                await RespondAsync(context, code is not null);
            }
        }
        if (code is null) throw new CloudException(error ?? "Sign-in was cancelled.");

        using JsonDocument session = await PostJsonAsync($"{_url}/auth/v1/token?grant_type=pkce",
            new { auth_code = code, code_verifier = verifier }, token: null, cancel);
        SaveSession(ParseSession(session.RootElement));
    }

    private static async Task RespondAsync(HttpListenerContext context, bool ok)
    {
        string html = "<!doctype html><html><head><meta charset='utf-8'><title>Decrypt's ClipShot</title>" +
                      "<style>body{background:#09090b;color:#f4f4f5;font-family:Segoe UI,sans-serif;display:flex;" +
                      "align-items:center;justify-content:center;height:100vh;margin:0}div{text-align:center}" +
                      "h1{font-size:22px}p{color:#8e8e96}</style></head><body><div>" +
                      (ok ? "<h1>You're signed in</h1><p>You can close this tab and go back to ClipShot.</p>"
                          : "<h1>Sign-in didn't finish</h1><p>Go back to ClipShot and try again.</p>") +
                      "</div></body></html>";
        byte[] bytes = Encoding.UTF8.GetBytes(html);
        context.Response.ContentType = "text/html; charset=utf-8";
        context.Response.ContentLength64 = bytes.Length;
        await context.Response.OutputStream.WriteAsync(bytes);
        context.Response.Close();
    }

    private static Session ParseSession(JsonElement root)
    {
        string access = root.GetProperty("access_token").GetString() ?? "";
        string refresh = root.GetProperty("refresh_token").GetString() ?? "";
        int expiresIn = root.TryGetProperty("expires_in", out JsonElement e) && e.TryGetInt32(out int seconds) ? seconds : 3600;
        JsonElement user = root.GetProperty("user");
        string id = user.GetProperty("id").GetString() ?? "";
        string name = "", avatar = "";
        if (user.TryGetProperty("user_metadata", out JsonElement meta))
        {
            foreach (string field in new[] { "name", "full_name", "user_name" })
                if (name.Length == 0 && meta.TryGetProperty(field, out JsonElement value)) name = value.GetString() ?? "";
            if (meta.TryGetProperty("avatar_url", out JsonElement picture)) avatar = picture.GetString() ?? "";
        }
        int hash = name.IndexOf('#');
        if (hash > 0) name = name[..hash];
        return new Session(access, refresh, DateTimeOffset.UtcNow.AddSeconds(expiresIn - 60), id,
                           name.Length > 0 ? name : "Player", avatar.Length > 0 ? avatar : null);
    }

    /// <summary>A valid access token, refreshed first if it's about to expire.</summary>
    private async Task<string> TokenAsync(CancellationToken cancel)
    {
        Session session = _session ?? throw new CloudException("Please sign in.");
        if (DateTimeOffset.UtcNow < session.ExpiresAt) return session.AccessToken;
        try
        {
            using JsonDocument refreshed = await PostJsonAsync($"{_url}/auth/v1/token?grant_type=refresh_token",
                new { refresh_token = session.RefreshToken }, token: null, cancel);
            Session next = ParseSession(refreshed.RootElement);
            SaveSession(next);
            return next.AccessToken;
        }
        catch (CloudException exception) when (exception.Status is 400 or 401)
        {
            // The sign-in was revoked or expired for good.
            SaveSession(null);
            throw new CloudException("Your sign-in expired. Please sign in again.", 401);
        }
    }

    /// <summary>Checks the saved sign-in with the server when online (offline is fine).</summary>
    public async Task<bool> RefreshIfPossibleAsync(CancellationToken cancel = default)
    {
        if (_session is null) return false;
        try
        {
            await TokenAsync(cancel);
            return true;
        }
        catch (CloudException exception) when (exception.Status == 401)
        {
            return false;
        }
        catch (Exception exception) when (exception is HttpRequestException or TaskCanceledException or CloudException)
        {
            return true; // offline or server busy: keep the saved sign-in
        }
    }

    // ---------- Friends ----------

    public sealed record Profile(string Id, string Username, string? DisplayName, string? AvatarUrl);
    public sealed record Friend(Profile Person, bool Accepted, bool IncomingRequest);

    public async Task<List<Friend>> GetFriendsAsync(CancellationToken cancel = default)
    {
        string me = UserId ?? "";
        using JsonDocument links = await RestAsync(HttpMethod.Get, "friendships?select=requester,addressee,status", null, cancel);
        var rows = links.RootElement.EnumerateArray().Select(row => (
            Requester: row.GetProperty("requester").GetString() ?? "",
            Addressee: row.GetProperty("addressee").GetString() ?? "",
            Accepted: row.GetProperty("status").GetString() == "accepted")).ToList();
        if (rows.Count == 0) return [];
        var ids = rows.Select(r => r.Requester == me ? r.Addressee : r.Requester).Distinct().ToList();
        Dictionary<string, Profile> people = await ProfilesAsync(ids, cancel);
        return rows
            .Select(r => (r, other: r.Requester == me ? r.Addressee : r.Requester))
            .Where(x => people.ContainsKey(x.other))
            .Select(x => new Friend(people[x.other], x.r.Accepted, !x.r.Accepted && x.r.Addressee == me))
            .OrderBy(f => f.Accepted).ThenBy(f => f.Person.Username, StringComparer.OrdinalIgnoreCase)
            .ToList();
    }

    private async Task<Dictionary<string, Profile>> ProfilesAsync(IEnumerable<string> ids, CancellationToken cancel)
    {
        string list = string.Join(",", ids);
        using JsonDocument found = await RestAsync(HttpMethod.Get, $"profiles?select=id,username,display_name,avatar_url&id=in.({list})", null, cancel);
        return found.RootElement.EnumerateArray().Select(ReadProfile).ToDictionary(p => p.Id);
    }

    private static Profile ReadProfile(JsonElement row) => new(
        row.GetProperty("id").GetString() ?? "",
        row.GetProperty("username").GetString() ?? "",
        row.TryGetProperty("display_name", out JsonElement d) ? d.GetString() : null,
        row.TryGetProperty("avatar_url", out JsonElement a) ? a.GetString() : null);

    /// <summary>Sends a friend request by Discord username (accepts theirs if they already asked you).</summary>
    public async Task<string> AddFriendAsync(string username, CancellationToken cancel = default)
    {
        string name = username.Trim().TrimStart('@');
        int hash = name.IndexOf('#');
        if (hash > 0) name = name[..hash];
        if (name.Length == 0) throw new CloudException("Type your friend's Discord username.");
        // ilike without wildcards = case-insensitive exact match.
        string escaped = name.Replace("\\", "\\\\").Replace("%", "\\%").Replace("_", "\\_");
        using JsonDocument found = await RestAsync(HttpMethod.Get,
            $"profiles?select=id,username,display_name,avatar_url&username=ilike.{Uri.EscapeDataString(escaped)}&limit=1", null, cancel);
        Profile? person = found.RootElement.EnumerateArray().Select(ReadProfile).FirstOrDefault();
        if (person is null)
            throw new CloudException($"No ClipShot user called \"{name}\". They need to sign in to ClipShot with Discord first.");
        if (person.Id == UserId) throw new CloudException("That's you!");

        List<Friend> current = await GetFriendsAsync(cancel);
        Friend? existing = current.FirstOrDefault(f => f.Person.Id == person.Id);
        if (existing is { Accepted: true }) return $"You and {person.Username} are already friends.";
        if (existing is { IncomingRequest: true })
        {
            await AcceptFriendAsync(person.Id, cancel);
            return $"You and {person.Username} are now friends.";
        }
        if (existing is not null) return $"Your request to {person.Username} is waiting for them to accept.";
        using JsonDocument _ = await RestAsync(HttpMethod.Post, "friendships",
            new { requester = UserId, addressee = person.Id }, cancel);
        return $"Friend request sent to {person.Username}.";
    }

    public async Task AcceptFriendAsync(string requesterId, CancellationToken cancel = default)
    {
        using JsonDocument _ = await RestAsync(HttpMethod.Patch,
            $"friendships?requester=eq.{requesterId}&addressee=eq.{UserId}", new { status = "accepted" }, cancel);
    }

    /// <summary>Declines a request, cancels yours, or unfriends.</summary>
    public async Task RemoveFriendAsync(string otherId, CancellationToken cancel = default)
    {
        string me = UserId ?? "";
        using JsonDocument _ = await RestAsync(HttpMethod.Delete,
            $"friendships?or=(and(requester.eq.{me},addressee.eq.{otherId}),and(requester.eq.{otherId},addressee.eq.{me}))", null, cancel);
    }

    public async Task<List<SharedClip>> GetFeedAsync(CancellationToken cancel = default)
    {
        using JsonDocument result = await FunctionAsync(new { action = "feed" }, cancel);
        return result.RootElement.GetProperty("clips").Deserialize<List<SharedClip>>() ?? [];
    }

    public async Task<string> GetVideoUrlAsync(string clipId, CancellationToken cancel = default)

        using JsonDocument result = await FunctionAsync(new { action = "view", clipId }, cancel);
        return result.RootElement.GetProperty("video_url").GetString() ?? "";

    public async Task DeleteSharedClipAsync(string clipId, CancellationToken cancel = default)

    public async Task DeleteAccountAsync(CancellationToken cancel = default)

    public sealed record ShareDetails(string Title, string Category, double? DistanceM, string? Weapon, string? Shot,
                                      string? Rarity, string? Killer, string? Victim, double? DurationS);

    /// <summary>Uploads an already-made share copy (video + thumbnail) and its details.</summary>
    public async Task ShareAsync(string videoPath, string? thumbnailPath, ShareDetails details,
                                 IProgress<double>? progress, CancellationToken cancel = default)
    {
        long size = new FileInfo(videoPath).Length;
        using JsonDocument created = await RestAsync(HttpMethod.Post, "shared_clips", new
        {
            title = Trim(details.Title, 120),
            category = Trim(details.Category, 40),
            distance_m = details.DistanceM,
            weapon = Trim(details.Weapon, 80),
            shot = Trim(details.Shot, 40),
            rarity = Trim(details.Rarity, 40),
            killer = Trim(details.Killer, 80),
            victim = Trim(details.Victim, 80),
            duration_s = details.DurationS
        }, cancel, returnRow: true);
        string clipId = created.RootElement[0].GetProperty("id").GetString() ?? "";
        try
        {
            using JsonDocument links = await FunctionAsync(new { action = "upload", clipId, size }, cancel);
            string videoUrl = links.RootElement.GetProperty("video_upload_url").GetString() ?? "";
            string thumbUrl = links.RootElement.GetProperty("thumbnail_upload_url").GetString() ?? "";
            if (thumbnailPath is not null && File.Exists(thumbnailPath))
                await UploadFileAsync(thumbUrl, thumbnailPath, "image/jpeg", null, cancel);
            await UploadFileAsync(videoUrl, videoPath, "video/mp4", progress, cancel);
            using JsonDocument _ = await FunctionAsync(new { action = "finish", clipId }, cancel);
        }
        catch
        {
            // Don't leave a half-shared clip behind.
            try { using JsonDocument _ = await FunctionAsync(new { action = "delete", clipId }, CancellationToken.None); }
            catch (Exception exception) when (exception is HttpRequestException or CloudException or TaskCanceledException) { }
            throw;
        }
    }

    private static string? Trim(string? text, int length) =>
        string.IsNullOrWhiteSpace(text) ? null : text.Length <= length ? text : text[..length];

    private static async Task UploadFileAsync(string url, string path, string type, IProgress<double>? progress, CancellationToken cancel)
    {
        await using FileStream file = File.OpenRead(path);
        using var content = new ProgressStreamContent(file, progress);
        content.Headers.ContentType = new MediaTypeHeaderValue(type);
        content.Headers.ContentLength = file.Length;
        using HttpResponseMessage response = await Http.PutAsync(url, content, cancel);
        if (!response.IsSuccessStatusCode)
            throw new CloudException("The upload didn't go through. Check your internet and try again.", (int)response.StatusCode);
    }

    // ---------- HTTP helpers ----------

    private async Task<JsonDocument> RestAsync(HttpMethod method, string path, object? body, CancellationToken cancel, bool returnRow = false)
    {
        string token = await TokenAsync(cancel);
        using var request = new HttpRequestMessage(method, $"{_url}/rest/v1/{path}");
        request.Headers.Add("apikey", _key);
        request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        request.Headers.Add("Prefer", returnRow ? "return=representation" : "return=minimal");
        if (body is not null)
            request.Content = new StringContent(JsonSerializer.Serialize(body), Encoding.UTF8, "application/json");
        return await SendAsync(request, cancel);
    }

    private async Task<JsonDocument> FunctionAsync(object body, CancellationToken cancel)
    {
        string token = await TokenAsync(cancel);
        using var request = new HttpRequestMessage(HttpMethod.Post, $"{_url}/functions/v1/clips");
        request.Headers.Add("apikey", _key);
        request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        request.Content = new StringContent(JsonSerializer.Serialize(body), Encoding.UTF8, "application/json");
        return await SendAsync(request, cancel);
    }

    private async Task<JsonDocument> PostJsonAsync(string url, object body, string? token, CancellationToken cancel)
    {
        using var request = new HttpRequestMessage(HttpMethod.Post, url);
        request.Headers.Add("apikey", _key);
        if (token is not null) request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        request.Content = new StringContent(JsonSerializer.Serialize(body), Encoding.UTF8, "application/json");
        return await SendAsync(request, cancel);
    }

    private static async Task<JsonDocument> SendAsync(HttpRequestMessage request, CancellationToken cancel)
    {
        using HttpResponseMessage response = await Http.SendAsync(request, cancel);
        string text = await response.Content.ReadAsStringAsync(cancel);
        if (!response.IsSuccessStatusCode)
        {
            string message = "Something went wrong. Try again in a moment.";
            try
            {
                using JsonDocument error = JsonDocument.Parse(text);
                foreach (string field in new[] { "error", "msg", "message", "error_description" })
                    if (error.RootElement.ValueKind == JsonValueKind.Object &&
                        error.RootElement.TryGetProperty(field, out JsonElement value) && value.ValueKind == JsonValueKind.String)
                    {
                        message = value.GetString() ?? message;
                        break;
                    }
            }
            catch (JsonException) { }
            throw new CloudException(message, (int)response.StatusCode);
        }
        return JsonDocument.Parse(text.Length == 0 ? "null" : text);
    }

    private static string Base64Url(byte[] bytes) =>
        Convert.ToBase64String(bytes).TrimEnd('=').Replace('+', '-').Replace('/', '_');
}

public sealed class CloudException(string message, int status = 0) : Exception(message)
{
    public int Status { get; } = status;
}

/// <summary>Upload body that reports how much has been sent.</summary>
internal sealed class ProgressStreamContent(Stream stream, IProgress<double>? progress) : HttpContent
{
    protected override async Task SerializeToStreamAsync(Stream target, TransportContext? context)
    {
        byte[] buffer = new byte[256 * 1024];
        long sent = 0, total = stream.Length;
        int read;
        while ((read = await stream.ReadAsync(buffer)) > 0)
        {
            await target.WriteAsync(buffer.AsMemory(0, read));
            sent += read;
            progress?.Report(total > 0 ? (double)sent / total : 1);
        }
    }

    protected override bool TryComputeLength(out long length)
    {
        length = stream.Length;
        return true;
    }
}
